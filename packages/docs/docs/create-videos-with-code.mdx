<script>
(() => {
  'use strict';

  /* ====== CONFIG TIMELINE ====== */
  const TIMELINE = [
    { id: 'scene-1', start: 0,    end: 10000 },
    { id: 'scene-2', start: 10000, end: 25000 },
    { id: 'scene-3', start: 25000, end: 35000 },
    { id: 'scene-4', start: 35000, end: 45000 },
  ];
  const TOTAL_MS = 45000;

  /* ====== DOM CACHE ====== */
  const stage       = document.getElementById('stage');
  const fill        = document.getElementById('timelineFill');
  const counterEl   = document.getElementById('counter');
  const panelReal   = document.getElementById('panel-real');
  const btnPlay     = document.getElementById('btnPlay');
  const btnRestart  = document.getElementById('btnRestart');
  const scenes      = TIMELINE.map(t => document.getElementById(t.id));

  /* ====== STATE ====== */
  let rafId = null;
  let startTime = 0;
  let elapsed = 0;
  let playing = false;
  let currentSceneIdx = -1;
  let counterStarted = false;

  /* ====== HELPERS ====== */
  const easeOutCubic = t => 1 - Math.pow(1 - t, 3);

  function formatRupiah(n) {
    // Format angka menjadi "Rp 250 Juta"
    if (n >= 1_000_000 && n % 1_000_000 === 0) {
      return `Rp ${n / 1_000_000} Juta`;
    }
    return new Intl.NumberFormat('id-ID', { maximumFractionDigits: 0 }).format(n);
  }

  function setScene(idx) {
    if (idx === currentSceneIdx) return;
    scenes.forEach((el, i) => el.classList.toggle('active', i === idx));
    currentSceneIdx = idx;

    // Reset efek per scene
    if (idx === 1) {          // Scene 2 — shake pada panel real
      panelReal.classList.remove('active-shake');
      // trigger reflow agar animation bisa retrigger
      void panelReal.offsetWidth;
      panelReal.classList.add('active-shake');
    }
    if (idx === 2) {          // Scene 3 — counter
      counterStarted = false;
      counterEl.textContent = '0';
    }
  }

  /* ====== COUNTER ANIMATION ====== */
  const COUNTER_DURATION = 1800; // ms
  const COUNTER_TARGET = 250_000_000;

  function tickCounter(sceneElapsed) {
    if (counterStarted) return;
    const local = sceneElapsed - 25000;
    if (local < 0) return;
    counterStarted = true;

    const t0 = performance.now();
    const step = now => {
      const p = Math.min((now - t0) / COUNTER_DURATION, 1);
      const val = Math.floor(easeOutCubic(p) * COUNTER_TARGET);
      counterEl.textContent = formatRupiah(val);
      if (p < 1) requestAnimationFrame(step);
      else counterEl.textContent = 'Rp 250 Juta';
    };
    requestAnimationFrame(step);
  }

  /* ====== LOOP ====== */
  function loop(now) {
    if (!playing) return;
    if (!startTime) startTime = now;
    elapsed = now - startTime;

    // progress
    const progress = Math.min(elapsed / TOTAL_MS, 1);
    fill.style.width = `${progress * 100}%`;

    // scene index
    const idx = TIMELINE.findIndex(t => elapsed >= t.start && elapsed < t.end);
    const safeIdx = idx === -1 ? TIMELINE.length - 1 : idx;
    setScene(safeIdx);

    // efek khusus per scene
    if (safeIdx === 2) tickCounter(elapsed);

    // selesai
    if (elapsed >= TOTAL_MS) {
      playing = false;
      btnPlay.textContent = '↻ Ulangi';
      return;
    }
    rafId = requestAnimationFrame(loop);
  }

  /* ====== CONTROLS ====== */
  function play() {
    if (playing) return;
    if (elapsed >= TOTAL_MS) return restart();
    playing = true;
    startTime = performance.now() - elapsed;
    btnPlay.textContent = '⏸ Pause';
    rafId = requestAnimationFrame(loop);
  }

  function pause() {
    playing = false;
    if (rafId) cancelAnimationFrame(rafId);
    btnPlay.textContent = '▶ Play';
  }

  function restart() {
    pause();
    elapsed = 0;
    startTime = 0;
    currentSceneIdx = -1;
    counterStarted = false;
    counterEl.textContent = '0';
    fill.style.width = '0%';
    setScene(0);
    play();
  }

  btnPlay.addEventListener('click', () => {
    if (playing) pause();
    else play();
  });
  btnRestart.addEventListener('click', restart);

  /* Spacebar = play/pause, R = restart */
  window.addEventListener('keydown', e => {
    if (e.code === 'Space') { e.preventDefault(); playing ? pause() : play(); }
    if (e.key === 'r' || e.key === 'R') restart();
  });

  /* ====== INIT ====== */
  setScene(0); // tampilkan scene 1 saat load

  // Auto-play bila browser mengizinkan
  window.addEventListener('load', () => setTimeout(play, 400));
})();
</script>
