_Part of the [`@remotion/shapes`](/docs/shapes) package._

Generates a polygon SVG path.

## Example

```tsx twoslash title="polygon.ts"
import { makePolygon } from "@remotion/shapes";

const { path, width, height, transformOrigin, instructions } = makePolygon({
  points: 5,
  radius: 80,
});

console.log(path); // M 76.08452130361229 0 L 152.16904260722458 55.278640450004204 L 123.10734148701015 144.72135954999578 L 29.06170112021445 144.72135954999578 L 0 55.27864045000422 L 76.08452130361229 0
console.log(width); // 152.16904260722458
console.log(height); // 144.72135954999578
console.log(transformOrigin); // '80 80'
console.log(instructions); // '[{type: "M"}, ...]'
```
