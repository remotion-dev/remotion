type Mood = "happy" | "sad" | "mad" | "shy";
type Kind = "rubor" | "lab" | "som" | "del" | "mas" | "heart" | "house" | "stairs";

const COL = { pink: "#ff7fa8", coral: "#ff8a3d" };

const Face: React.FC<{ m: Mood; cx: number; ey: number; my: number; d: number }> = ({ m, cx, ey, my, d }) => (
  <g>
    <circle cx={cx - d} cy={ey} r={3.5} fill="#222" />
    <circle cx={cx + d} cy={ey} r={3.5} fill="#222" />
    {m === "happy" && <path d={`M${cx - 8} ${my} Q${cx} ${my + 10} ${cx + 8} ${my}`} {...line} />}
    {m === "sad" && (
      <>
        <path d={`M${cx - 7} ${my + 5} Q${cx} ${my - 3} ${cx + 7} ${my + 5}`} {...line} />
        <circle cx={cx + d + 3} cy={ey + 9} r={2.3} fill="#5bb8ff" />
      </>
    )}
    {m === "mad" && (
      <>
        <path d={`M${cx - d - 6} ${ey - 9} L${cx - d + 4} ${ey - 4} M${cx + d + 6} ${ey - 9} L${cx + d - 4} ${ey - 4}`} {...line} />
        <path d={`M${cx - 7} ${my + 4} Q${cx} ${my - 2} ${cx + 7} ${my + 4}`} {...line} />
      </>
    )}
    {m === "shy" && (
      <>
        <circle cx={cx - d - 5} cy={ey + 7} r={5} fill="#ff4d7a" opacity={0.5} />
        <circle cx={cx + d + 5} cy={ey + 7} r={5} fill="#ff4d7a" opacity={0.5} />
        <path d={`M${cx - 4} ${my + 1} Q${cx} ${my - 2} ${cx + 4} ${my + 1}`} {...line} />
      </>
    )}
  </g>
);

/* ============ ANIMACIÓN DE ESCENAS ============ */
const ActorView: React.FC<{ a: Actor }> = ({ a }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const from = a.from ?? 0;
  if (frame < from || (a.until !== undefined && frame >= a.until)) return null;
  const f = frame - from;

  let tx = 0, ty = 0, rot = 0, sc = 1;
  switch (a.anim) {
    case "bounce":
      ty = -Math.abs(Math.sin(frame / 8)) * 50;
      break;
    case "shake":
      rot = Math.sin(frame / 2) * 6;
      tx = Math.sin(frame / 2) * 10;
      break;
    case "fall":
      ty = interpolate(f, [0, 60], [0, 700], { extrapolateRight: "clamp", easing: Easing.in(Easing.quad) });
      rot = interpolate(f, [0, 60], [0, 140], { extrapolateRight: "clamp" });
      break;
    case "run":
      tx = interpolate(f, [0, 150], [0, 650], { extrapolateRight: "clamp" });
      ty = -Math.abs(Math.sin(f / 4)) * 25;
      break;
    case "pop":
      sc = spring({ frame: f, fps, config: { damping: 10 } });
      break;
  }

  return (
    <div
      style={{
        position: "absolute", left: `${a.x}%`, top: `${a.y}%`,
        transform: `translate(-50%, -50%) translate(${tx}px, ${ty}px) rotate(${rot}deg) scale(${sc})`,
      }}
    >
      <Character kind={a.kind} color={a.color} mood={a.mood} broken={a.broken} size={a.size ?? 220} />
    </div>
  );
};

const BubbleView: React.FC<{ b: Bubble }> = ({ b }) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();
  const from = b.from ?? 0;
  if (frame < from) return null;
  const sc = spring({ frame: frame - from, fps, config: { damping: 10 } });
  return (
    <div
      style={{
        position: "absolute", left: `${b.x}%`, top: `${b.y}%`,
        transform: `translate(-50%, -50%) scale(${sc})`,
        background: b.big ? "#ffe14d" : "#fff", color: "#4a1030",
        padding: b.big ? "24px 48px" : "16px 28px", borderRadius: 36,
        fontSize: b.big ? 110 : 48, fontWeight: 800, maxWidth: 520,
        textAlign: "center", boxShadow: "0 8px 20px rgba(0,0,0,.25)",
      }}
    >
      {b.text}
    </div>
  );
};

const Scene: React.FC<{ data: SceneData }> = ({ data }) => {
  const frame = useCurrentFrame();
  const { durationInFrames } = useVideoConfig();

  const opacity = interpolate(frame, [0, 10, durationInFrames - 10, durationInFrames], [0, 1, 1, 0]);
  const shake =
    data.shakeAt !== undefined && frame >= data.shakeAt && frame < data.shakeAt + 20
      ? Math.sin(frame * 2) * 18 : 0;

  const perLine = durationInFrames / data.lines.length;
  const ln = data.lines[Math.min(data.lines.length - 1, Math.floor(frame / perLine))];

  return (
    <AbsoluteFill style={{ opacity }}>
      <AbsoluteFill style={{ transform: `translate(${shake}px, ${shake / 2}px)` }}>
        {data.actors.map((a, i) => <ActorView key={i} a={a} />)}
        {data.bubbles?.map((b, i) => <BubbleView key={i} b={b} />)}
      </AbsoluteFill>

      <div
        style={{
          position: "absolute", left: 60, right: 60, bottom: 140,
          background: "rgba(255,255,255,.92)", color: "#4a1030", borderRadius: 36,
          padding: "28px 36px", textAlign: "center", boxShadow: "0 8px 24px rgba(0,0,0,.2)",
        }}
      >
        <div style={{ fontSize: 34, fontWeight: 700, color: "#e0457b", marginBottom: 8 }}>{ln.who}</div>
        <div style={{ fontSize: 46, fontWeight: 600, lineHeight: 1.25 }}>{ln.text}</div>
      </div>
    </AbsoluteFill>
  );
};

        return (

      })}

  );
};
