<script>
const scenes = [
  ["🐰","Once upon a time, a little bunny lived in a magical forest..."],
  ["🐰","One night, he discovered a tiny glowing star! ⭐"],
  ["🐰","'Oh no! You belong in the sky!' whispered the bunny."],
  ["🐰","He gently helped the little star return to the sky."],
  ["🐰","The star shone brightly, just for its little friend! ✨"],
  ["🐰","A little kindness can make a big difference. ❤️"]
];

let timer;

function startStory() {
  clearInterval(timer);
  let i = 0;
  document.querySelector("button").style.display = "none";

  function showScene() {
    if (i >= scenes.length) {
      clearInterval(timer);
      document.querySelector("button").style.display = "block";
      document.querySelector("button").textContent = "↻ Replay";
      return;
    }
    document.getElementById("bunny").textContent = scenes[i][0];
    document.getElementById("story").textContent = scenes[i][1];
    i++;
  }

  showScene();
  timer = setInterval(showScene, 2500);
}
</script>
